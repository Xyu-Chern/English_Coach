def 下载任务(conn):
    for 集号, 网址 in 取待下载单集(conn):
        正文 = 下载字幕(网址)
        with conn.transaction():
            conn.execute(
                "INSERT INTO 原文表 (集号, 正文) VALUES (%s, %s)",
                (集号, 正文))

                "UPDATE 任务表 SET 已下载 = true WHERE 集号 = %s",

原文表 → 小段表

execute 发出 SELECT，fetchall 取回行。切段函数交回“起点、终点、小段正文”，位置是原文里的字符位置。

def 切段任务(conn):
    文档 = conn.execute(
        "SELECT 集号, 正文 FROM 原文表 WHERE 已切段 = false"
    ).fetchall()
    for 集号, 正文 in 文档:
        with conn.transaction():
            for 起点, 终点, 片段 in 解析并切段(正文):
                conn.execute(
                    "INSERT INTO 小段表 (集号, 起点, 终点, 正文) "
                    "VALUES (%s, %s, %s, %s)",
                    (集号, 起点, 终点, 片段))
            conn.execute(
                "UPDATE 原文表 SET 已切段 = true WHERE 集号 = %s",
                (集号,))

小段表 → 模型 → 向量表

按段号查哪些小段还缺向量，只把正文发给模型；保存时用同一个段号把数字和原文对上。

def 向量任务(conn, 模型):
    待处理 = conn.execute("""
        SELECT 段号, 正文 FROM 小段表
        WHERE 段号 NOT IN (SELECT 段号 FROM 向量表)
        ORDER BY 段号 LIMIT 32
    """).fetchall()
    if not 待处理:
        return
    文本 = [正文 for 段号, 正文 in 待处理]
    一批向量 = 模型.embed_texts(文本).vectors
    with conn.transaction():
        for (段号, _), 向量 in zip(待处理, 一批向量, strict=True):
            conn.execute(
                "INSERT INTO 向量表 (段号, 数字) VALUES (%s, %s::vector)",
                (段号, json.dumps(向量)))

假设其中一段的段号是 8，向量表就以 8 关联它；下次查询会跳过这段。　项目里的存储代码 ↗

最后从 main() 走一遍

教学版把三个任务顺序调用一次，同一个 conn 传到底。读代码时从这里出发。

import json
import psycopg

def main():
    with psycopg.connect(数据库地址, autocommit=True) as conn:
        下载任务(conn)                 # 网站 → 原文表
        切段任务(conn)                 # 原文表 → 小段表
        if 向量化已开启:
            向量任务(conn, 模型)       # 小段表 → 向量表

if __name__ == "__main__":
    main()

项目实际部署时，cron 在三个时间点分别启动三个任务。各任务各自连接同一个数据库，通过已保存的记录交接。
